create domain note.parts_type as text
    check (value in ('jpeg', 'png', 'text', 'tex', 'md', 'binary', 'url'))
;

-- フォルダ管理
--
-- deleted_number
--     値0     = 削除されていない状態
--     値1以上 = 削除されている状態
-- 削除処理
--     aid, parent, nameでのdeleted_number最大値+1の値で、deleted_numberを更新する
create table note.folder (
    id             serial  primary key,        -- フォルダID
    aid            integer not null,           -- アカウントID
    parent         integer null,               -- 親フォルダのフォルダID
    name           text    not null,           -- フォルダ名称
    dorder         integer not null,           -- 表示順
    
    deleted_number integer not null default 0, -- 削除番号

    -- アカウントID 外部キー
    constraint fk_note_folder_aid
        foreign key (aid)
        references accounts(id),

    -- 親フォルダ 外部キー
    constraint fk_note_folder_parent
        foreign key (parent)
        references note.folder(id),

    -- フォルダ名称 唯一性
    constraint uq_note_folder_name
        unique (aid, parent, deleted_number, name),

    -- 表示順 唯一性
    constraint uq_note_folder_dorder
        unique (aid, parent, dorder)
);

-- ファイル管理
--
-- deleted_number
--     値0     = 削除されていない状態
--     値1以上 = 削除されている状態
-- 削除処理
--     aid, parent, titleでのdeleted_number最大値+1の値で、deleted_numberを更新する
create table note.file (
    id             serial   primary key,       -- ファイルID
    aid            integer  not null,          -- アカウントID
    
    belong         integer not null,           -- 所属するフォルダID
    title          text    not null,           -- ファイルタイトル
    dorder         integer not null,           -- 表示順
    
    deleted_number integer not null default 0, -- 削除番号

    -- アカウントID 外部キー
    constraint fk_note_file_aid
        foreign key (aid)
        references accounts(id),

    -- 所属フォルダ 外部キー
    constraint fk_note_file_belong
        foreign key (belong)
        references note.folder(id),

    -- タイトル 唯一性
    constraint uq_note_file_title
        unique (aid, belong, deleted_number, title),

    -- 表示順 唯一性
    constraint uq_note_file_dorder
        unique (aid, belong, dorder)
);

-- ファイルパーツ
create table note.parts (
    id  serial primary key,                   -- パーツID
    aid integer not null,                     -- アカウントID

    file       integer not null,              -- 所属するファイルのファイルID
    dorder     integer not null,              -- 表示順
    is_deleted bool not null,                 -- 削除フラグ

    ptype      note.parts_type not null,
    data       text            not null,

    constraint fk_note_parts_aid
        foreign key (aid)
        references accounts(id),
    constraint fk_note_parts_file
        foreign key (file)
        references note.file(id),
    constraint uq_note_parts_dorder
        unique (aid, file, dorder)
);

共通事項
  JWTのusername=accounts.usernameを条件に、取得したaccounts.idをアカウントIDとする。
  JWTの詳細はJWT_USERNAME_TECH_SPEC.mdを参照のこと

  追加や更新対象のレコードは、自身のアカウントIDのレコードのみとする。
  該当レコードが、自身のアカウントIDに無い場合には処理異常とする。

A-1. アイテム一覧取得
  処理概要
    指定したフォルダに所属するフォルダ、ファイルの一覧を返す
    削除済みを対象とするのか、対象から外すのかの指定を可能とする

  Input
    {
      folder_id: N,
      include_deleted: true/false
    }
    
    folder_id: 対象フォルダID
    include_deleted: 削除済み対象フラグ

  Output
    {
      parent: {
        id: n0,
        name: nm0
      },
      folder: [
        { id: fld_n1, dorder: fld_o1, name: fld_nm1, is_del: true/false },
        ...
      ],
      file: [
        { id: fil_n1, dorder: fil_o1, title: fil_nm1, is_del: true/false },
        ...
      ]
    }
    
    parent.id: 対象フォルダID
    parent.name: 対象フォルダのフォルダ名称
    folder[n].id: フォルダID
    folder[n].dorder: フォルダの表示順
    folder[n].name: フォルダ名称
    folder[n].is_del: 削除されたものであればtrue
    file[n].id: ファイルID
    file[n].dorder: ファイルの標示順
    file[n].title: ファイル名称
    file[n].is_del: 削除されたものであればtrue

  処理内容
    アカウントIDを取得する
    note.folder、note.fileを検索し、対象のものを検索し、応答MSGを送信する。。


A-2. ファイル取得
  処理概要
    指定されたファイルの情報を検索して返す。

  Input
    {
      file_id: N
    }

    file_id: 対象ファイルID

  Output
    {
      id: N,
      belong: {
        id: M,
        name: xxx
      },
      title: yyy,
      parts: [
        { id: n1, dorder: d1, ptype: p1, data: dt1 },
        ...
      ]
    }

  処理内容
    アカウントIDを取得する
    note.folder、note.file、note.partsを検索する。

B-1. フォルダ作成
  処理概要
    フォルダを作成する
  
  Input
    {
      parent_id: N,
      name: aaa
    }
  
  Output
    {
      result: true/false
      reason: 異常理由
    }
  
  処理内容
    アカウントIDを取得する。
    note.folderにレコードを追加する。
    dorderには、同一parentでの最大値+1を使用する。
    deleted_numberは0を使用する。
    同じアカウントIDで、同一parentで、deleted_number=0で、同一nameのレコードが既に存在している場合には、エラーを返す。

B-2. フォルダ削除
  処理概要
    指定されたフォルダを削除状態にする
  
  Input
    {
      folder_id: n
    }
  
  Output
    {
      result: true/false
      reason: 異常理由
    }
  
  処理内容
    アカウントIDを取得する。
    note.folderのdeleted_numberを更新する。
    aid、parent、nameで、既に存在するdeleted_numberの値+1で、更新する。

B-3. フォルダ削除解除
  処理概要
    指定されたフォルダの削除状態を解除する
  
  Input
    {
      folder_id: n
    }
  
  Output
    {
      result: true/false
      reason: 異常理由
    }
  
  処理内容
    アカウントIDを取得する。
    note.folderのdeleted_numberを0で更新する。
    aid、parent、nameで、既にdeleted_number=0のレコードが存在する場合には、処理異常とする。
